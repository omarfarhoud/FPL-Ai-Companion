// Find players with incorrect Defender position assignments
// A player has an incorrect Defender position if their actual score doesn't match
// what they would have scored as a Defender according to FPL scoring rules

MATCH (p:Player)-[:PLAYS_AS]->(pos:Position)
WITH p, collect(pos.name) AS positions
WHERE size(positions) = 2 AND 'DEF' IN positions

MATCH (p)-[played:PLAYED_IN]->(f:Fixture)
MATCH (f)<-[:HAS_FIXTURE]-(gw:Gameweek)<-[:HAS_GW]-(s:Season)

// Calculate what the score would be if player was a Defender
WITH p, s, f, played,
  (
    // General scoring
    COALESCE(played.assists, 0) * 3 +
    // Minutes: < 60 mins = 1 pt, >= 60 mins = 2 pts (if mins > 0)
    CASE 
      WHEN played.minutes > 0 AND played.minutes < 60 THEN 1
      WHEN played.minutes >= 60 THEN 2
      ELSE 0
    END +
    COALESCE(played.penalties_missed, 0) * -2 +
    COALESCE(played.own_goals, 0) * -2 +
    COALESCE(played.yellow_cards, 0) * -1 +
    COALESCE(played.red_cards, 0) * -3 +
    // *** CRITICAL ADDITION: Bonus points are part of total_points ***
    COALESCE(played.bonus, 0) + 

    // Defender-specific scoring
    COALESCE(played.goals_scored, 0) * 6 +  
    // Clean sheets (only if played > 60 mins)
    CASE 
      WHEN played.minutes >= 60 AND COALESCE(played.clean_sheets, 0) > 0 THEN 4
      ELSE 0
    END +
    // Goals conceded (-1 per 2 goals)
    toInteger(floor(toFloat(COALESCE(played.goals_conceded, 0)) / 2.0)) * -1
  ) AS def_score

// Check if defender score matches actual score
WITH p, s, f, played, def_score,
  CASE 
    WHEN def_score = played.total_points THEN true
    ELSE false
  END AS matches_def

// Count total fixtures and incorrect fixtures per player
WITH p.player_name AS player,
  count(f) AS total_fixtures,
  sum(CASE WHEN NOT matches_def THEN 1 ELSE 0 END) AS not_def_fixtures,
  collect(
    CASE 
      WHEN NOT matches_def THEN {
        def_score: def_score,
        season: s.season_name,
        fixture_number: f.fixture_number,
        matches_def: matches_def,
        total_points: played.total_points
      }
      ELSE null
    END
  ) AS all_fixtures

// Filter out null values and return ALL suspects (even if 0 errors)
WITH player, total_fixtures, not_def_fixtures,
  [fixture IN all_fixtures WHERE fixture IS NOT null] AS fixture_details

// REMOVED: WHERE not_def_fixtures > 0 
// This ensures Stones and Azpi show up with 0 errors.

RETURN player, not_def_fixtures, total_fixtures, fixture_details
ORDER BY player
